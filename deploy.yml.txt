
name: Publicar NIVEST

on:
  workflow_dispatch:
  push:
    paths:
      - 'NIVEST-projeto-inicial.zip'
      - '.github/workflows/deploy.yml'

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Baixar repositorio
        uses: actions/checkout@v4

      - name: Configurar Node
        uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Extrair projeto
        run: |
          unzip -q NIVEST-projeto-inicial.zip -d source
          test -f source/nivest-app/package.json

      - name: Construir NIVEST
        working-directory: source/nivest-app
        run: |
          npm install
          printf '%s\n' '/// <reference types="vite/client" />' > src/vite-env.d.ts
          npm run build

      - name: Configurar Pages
        uses: actions/configure-pages@v5

      - name: Preparar publicacao
        uses: actions/upload-pages-artifact@v3
        with:
          path: source/nivest-app/dist

  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Publicar
        id: deployment
        uses: actions/deploy-pages@v4
